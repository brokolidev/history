#! /usr/bin/python2.7

import smtplib

from email.MIMEMultipart import MIMEMultipart

from email.MIMEBase import MIMEBase

from email.MIMEText import MIMEText

import os

import platform

import datetime

import socket

# get the basic information

hostname = platform.node()

type = platform.platform()

dateinfo = os.stat( "/root/install.log" )

installed_date =  datetime.date.fromtimestamp( dateinfo.st_ctime )

# sending mail function

def send_gmail( to, subject, text ):

msg = MIMEMultipart()

msg['From'] = "보내는 사람 주소"

msg['To'] = to

msg['Subject'] = subject

msg.attach( MIMEText( text) )

mailServer = smtplib.SMTP( "smtp.gmail.com", 587 )

mailServer.ehlo()

mailServer.starttls()

mailServer.ehlo()

mailServer.login( 구글계정, 구글계정비번 )

mailServer.sendmail( 구글계정, to, msg.as_string() )

mailServer.close()

# get ip address function

def get_localip( host ):

s = socket.socket( socket.AF_INET, socket.SOCK_DGRAM )

try:

s.connect(( host, 9 ) )

client = s.getsockname()[0]

except socket.error:

client = "Unknown IP"

finally:

del s

return client

body_text = "Hostname : " + hostname + " \n" +\

"IP Address : " + get_localip( 통신을 할 IP주소 ) + "\n" +\

"OS Version : " + type + "\n" +\

"Installed Date : " + str( installed_date )

send_gmail( 수신 Email 주소, 제목, body_text )
