```javascript
// 숫자만 적용 ex) 0123456789
var regNum = /^[0-9]+$/;

// 전화번호 형식만 적용 ex) 123-456-7890, 01-234-5678, 123-4567-8901
var regPhone = /^[0-9]{2,3}-[0-9]{3,4}-[0-9]{4}$/;

// 메일 형식만 적용 ex) abc@abc.abc
var regMail = /^[_a-zA-Z0-9-]+@[._a-zA-Z0-9-]+.[a-zA-Z]+$/;

// 도메인 형식만 적용 ex) abcd.efc
var regDomain = /^[.a-zA-Z0-9-]+.[a-zA-Z]+$/;

// 영문만 적용 ex) abc, ABC
var regAlpha = /^[a-zA-Z]+$/;

// 한글만 적용(첫자리만)
var regHangul = /[가-힣]/;

// 한글과 영문만 적용(첫자리만)
var regHangulEng = /[가-힣a-zA-Z]/;

// 한글만 적용(자리에 상관없이)
var regHangulOnly = /^[가-힣]*$/;

// 아이디 적용 (영문대소문자, 숫자, _ , -  사용가능, 4~19자리)
var regId = /^[a-zA-Z]{1}[a-zA-Z0-9_-]{4,19}$/;
```

적용 시키는 방법은 담은 변수에 test 메서드를 이용한다.

test 메서드는 정규식 조건에 부합하면 true, 아니면 false 를 반환한다.

ex)

```javascript
if (!regPhone.test('aaa')) {
    alert("전화번호 형식과 맞지 않습니다.");
}
```
