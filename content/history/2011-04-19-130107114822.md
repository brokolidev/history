이메일 체크 정규식
/^[0-9a-zA-Z]([-_\.]?[0-9a-zA-Z])*@[0-9a-zA-Z]([-_\.]?[0-9a-zA-Z])*\.[a-zA-Z]{2,3}$/i;

핸드폰번호 정규식
/^\d{3}-\d{3,4}-\d{4}$/;

일반 전화번호 정규식
/^\d{2,3}-\d{3,4}-\d{4}$/;

아이디나 비밀번호 정규식
/^[a-z0-9_]{4,20}$/;

var regExp = /^01([0|1|6|7|8|9]?)-?([0-9]{3,4})-?([0-9]{4})$/;
