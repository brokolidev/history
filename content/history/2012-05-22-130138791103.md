smtplib 및 email 라이브러리를 통해서 메일 전송이 가능하다.

```python
#!/usr/bin/python
import smtplib
from email.MIMEMultipart import MIMEMultipart
from email.MIMEBase import MIMEBase
from email.MIMEText import MIMEText
import os

user = "자기 이메일 주소"
pswd = "해당 이메일 패스워드"

def send_gmail(to, subject, text):
    msg = MIMEMultipart()
    msg['From'] = user
    msg['To'] = to
    msg['Subject'] = subject
    msg.attach(MIMEText(text))

    mailServer = smtplib.SMTP("smtp.gmail.com", 587)
    mailServer.ehlo()
    mailServer.starttls()
    mailServer.ehlo()
    mailServer.login(user, pswd)
    mailServer.sendmail(user, to, msg.as_string())
    mailServer.close()
```
