음 Replace를 하지 않으니 match 를 써서 간단하게 구현.

var filter = ['test1, 'test2', 'test3'];

var matchcnt = 0;

$('a.submit').live('click', function(e){

matchcnt = 0; // 카운트 초기화

var txt = $('textarea').val(); // Textarea에서 내용을 가져온다. jquery 이용.

for( var i in filter ){

try{

var compare = txt.match( filter[i] );

console.log( compare.index );

alert( '금지어가 포함 되어있습니다. - ' + filter[i] );

matchcnt++;

if( matchcnt > 0 ) return;

} catch( err ) {

console.log( '통과' );

}

}

});
